boxscore.players[].stats array indexes:
[0] = minutes
[1] = points
[2] = FG (made-attempted, e.g., "7-10")
[3] = 3PT (made-attempted)
[4] = FT (made-attempted)
[5] = rebounds total
[6] = assists
[7] = turnovers
[8] = steals
[9] = blocks
[10] = offensive rebounds
[11] = defensive rebounds
[12] = personal fouls
[13] = plus-minus
